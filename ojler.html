function [x, y] = ojlerova_metoda(f, a, p, b, n)
% Parametri:
% f : funkcija koja predstavlja desnu stranu jednačine y' = f(x, y)
% a : početak intervala
% p : početni uslov y(a) = p
% b : kraj intervala
% n : ukupan broj tačaka (indeksirane od 1 do n)
 
% Računanje koraka integracije na osnovu n tačaka na intervalu [a, b]
h = (b - a) / (n - 1);
% Inicijalizacija vektora dužine n
x = linspace(a, b, n);
y = zeros(1, n);
y(1) = p;
% Iterativno računanje vrijednosti
for i = 1:(n - 1)
y(i+1) = y(i) + h * f(x(i), y(i));
end
end

S obzirom na to da je ovo višekoračna metoda, za izračunavanje vrijednosti neophodno je poznavati vrijednosti iz dva prethodna čvora ( i ). Na samom početku () poznajemo samo početni uslov , pa nam nedostaje vrijednost u čvoru unazad. Zbog toga se prva pomoćna vrijednost mora izračunati pomoću obične Ojlerove metode: , nakon čega se za sve ostale korake primjenjuje glavna formula.
function [x, y] = poboljsana_ojlerova_metoda(f, a, p, b, n)
% Parametri:
% f : funkcija koja predstavlja desnu stranu jednačine y' = f(x, y)
% a : početak intervala
% p : početni uslov y(a) = p
% b : kraj intervala
% n : ukupan broj tačaka (indeksirane od 1 do n)
 
% Računanje koraka integracije
h = (b - a) / (n - 1);
% Inicijalizacija vektora
x = linspace(a, b, n);
y = zeros(1, n);
y(1) = p;
% Prvi korak se mora uraditi običnim Ojlerom (jer nemamo y(0) u kodu)
y(2) = y(1) + h * f(x(1), y(1));
% Ostali koraci idu preko Milnove formule iz priloženog dokumenta
for i = 2:(n - 1)
y(i+1) = y(i-1) + 2 * h * f(x(i), y(i));
end
end
